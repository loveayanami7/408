#### 存储方式
1. 邻接矩阵
	1. 优点：
		1. 便于判断两个顶点之间是否有边
		2. 便于计算各个顶点的度
	2. 缺点：
		1. 不便于增加和删除节点
		2. 不便于统计边的数目，需要遍历邻接矩阵才能统计
		3. 空间复杂度高，只适合稠密图，稀疏图会造成空间的浪费
2. 邻接表
	1. 优点：
		1. 便于增加和删除节点
		2. 便于统计边的数目
		3. 空间效率高，适合稀疏图
	2. 缺点：
		1. 不便于判断顶点间是否有边
		2. 不便于计算有向图各顶点的度
3. 十字链表：存有向图
4. 邻接多重表：存无向图

bool visited[MAX_VERTEX_NUM];
void BFSTraverse(Graph G) {
	for(i=0; i<G.vexnum; i++)
		visited[i]=FALSE;
	InitQueue(Q);
	for(i=0; i<G.vexnum; i++)
		if(!visited[i])
			BFS(G,i);
}

// 基于邻接表
void BFS(ALGraph G, int i) {
	visit(i);
	visited[i]=TRUE;
	EnQueue(Q,i);
	while(!QueueEmpty(Q)){
		DeQueue(Q,v);
		for(p=G.vertices[v].firstarc;p;p=p->nextarc){
			w = p->adjvex;
			if(visited[w]==FALSE) {
				visit(w);
				visited[w]=TRUE;
				EnQueue(Q,w);
			}
		}
	}
}

// 基于邻接矩阵
void BFS(MGraph G, int i) {
	visit(i);
	visited[i]=TRUE;
	EnQueue(Q,i);
	while(!QueueEmpty(Q)) {
		DeQueue(Q,v);
		for(w=0;w<G.vexnum;w++) {
			if(visited[w]==FALSE && G.edge[v][w]==1) {
				visit(w);
				visited[w]=TRUE;
				EnQueue(Q,w);
			}
		}
	}
}

// BFS求解单源最短路径
void BFS_MIN_DISTANCE(Graph G, int u) {
	for(int i=0; i<G.vexnum; i++) {
		d[i] = INT_MAX;
	}
	visited[u]=TRUE;
	d[u]=0;
	EnQueue(Q,u);
	while(!QueueEmpty(Q)){
		DeQueue(Q,v);
		for(w=FirstNeighbor(G,u); w>=0; w=NextNeighbor(G,u,w)) {
			if(!visited[w]) {
				visited[w] = TRUE;
				d[w] = d[u]+1;
				EnQueue(Q,w);
			}
		}
	}
}
